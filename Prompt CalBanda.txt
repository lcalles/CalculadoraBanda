// --- Utilidades ---
function roundLinks(widthIn, incIn, mode) {
  const x = widthIn / incIn;
  if (mode === "floor") return Math.floor(x);
  if (mode === "ceil") return Math.ceil(x);
  return Math.round(x);
}

function combosSumTarget(target, partSizes, limit = 1200) {
  const sizes = [...new Set(partSizes)].sort((a, b) => b - a);
  const memo = new Map();
  function key(rem, idx) { return rem + ":" + idx; }
  function rec(rem, idx) {
    const k = key(rem, idx);
    if (memo.has(k)) return memo.get(k);
    if (rem === 0) return [[]];
    if (rem < 0 || idx >= sizes.length) return [];
    const s = sizes[idx];
    const maxK = Math.floor(rem / s);
    const out = [];
    for (let kcount = maxK; kcount >= 0; kcount--) {
      const part = Array(kcount).fill(s);
      const tails = rec(rem - s * kcount, idx + 1);
      for (const t of tails) {
        out.push([...part, ...t]);
        if (out.length >= limit) break;
      }
      if (out.length >= limit) break;
    }
    memo.set(k, out);
    return out;
  }
  const res = rec(target, 0);
  res.sort((a, b) => {
    // menos piezas; y favorecer piezas grandes
    if (a.length !== b.length) return a.length - b.length;
    for (let i = 0; i < Math.max(a.length, b.length); i++) {
      const ai = a[i] ?? 0, bi = b[i] ?? 0;
      if (ai !== bi) return bi - ai;
    }
    return 0;
  });
  return res.slice(0, limit);
}

function seamPositionsFromSegments(segments) {
  const seams = new Set();
  let acc = 0;
  for (let i = 0; i < segments.length; i++) {
    acc += segments[i].size;
    if (i !== segments.length - 1) seams.add(acc); // no añadir borde final
  }
  return seams;
}

function makeRowPatterns(widthLinks, edgeSizes, internalSizes, maxPatterns = 500) {
  const patterns = [];
  const eSizes = [...new Set(edgeSizes)].sort((a, b) => b - a);
  const iSizes = [...new Set(internalSizes)].sort((a, b) => b - a);

  for (const le of eSizes) {
    for (const re of eSizes) {
      if (le + re > widthLinks) continue;
      const remain = widthLinks - le - re;
      if (remain === 0) {
        const segs = [
          { role: "L", size: le, key: `edgeL_${le}` },
          { role: "R", size: re, key: `edgeR_${re}` },
        ];
        const seams = Array.from(seamPositionsFromSegments(segs));
        const pieceCount = 2;
        const score2 = -(le * le + re * re);
        patterns.push({ segments: segs, seams, pieceCount, score: [pieceCount, score2], widthLinks });
        continue;
      }
      const combos = combosSumTarget(remain, iSizes, 1200);
      for (const combo of combos) {
        const base = combo.slice();
        const segs = [
          { role: "L", size: le, key: `edgeL_${le}` },
          ...base.map((w, idx) => ({ role: "I", size: w, key: `int_${w}_${idx}` })),
          { role: "R", size: re, key: `edgeR_${re}` },
        ];
        const seams = Array.from(seamPositionsFromSegments(segs));
        const pieceCount = 2 + base.length;
        const score2 = -[...base, le, re].reduce((s, x) => s + x * x, 0);
        patterns.push({ segments: segs, seams, pieceCount, score: [pieceCount, score2], widthLinks });

        // Variantes por rotación de internos (patrón ladrillo)
        for (let shift = 1; shift < Math.min(4, base.length); shift++) {
          const rotated = base.slice(shift).concat(base.slice(0, shift));
          const segs2 = [
            { role: "L", size: le, key: `edgeL_${le}` },
            ...rotated.map((w, idx) => ({ role: "I", size: w, key: `int_${w}_rot${shift}_${idx}` })),
            { role: "R", size: re, key: `edgeR_${re}` },
          ];
          const seams2 = Array.from(seamPositionsFromSegments(segs2));
          const pieceCount2 = 2 + rotated.length;
          const score22 = -[...rotated, le, re].reduce((s, x) => s + x * x, 0);
          patterns.push({ segments: segs2, seams: seams2, pieceCount: pieceCount2, score: [pieceCount2, score22], widthLinks });
        }
      }
    }
  }

  // Deduplicar por conjunto de juntas + número de piezas
  const dedup = new Map();
  for (const p of patterns) {
    const k = JSON.stringify({ s: [...p.seams].sort((a,b)=>a-b), c: p.pieceCount });
    if (!dedup.has(k) || (p.score[0] < dedup.get(k).score[0]) || (p.score[0] === dedup.get(k).score[0] && p.score[1] < dedup.get(k).score[1])) {
      dedup.set(k, p);
    }
  }
  const uniq = Array.from(dedup.values());
  uniq.sort((a, b) => (a.pieceCount - b.pieceCount) || (a.score[1] - b.score[1]));
  return uniq.slice(0, maxPatterns);
}

function layoutRows(numRows, patterns) {
  if (!patterns.length) return [];
  const chosen = [];
  let prevSeams = new Set();
  for (let i = 0; i < numRows; i++) {
    let picked = null;
    for (const p of patterns) {
      const inter = p.seams.filter((x) => prevSeams.has(x));
      if (inter.length === 0) { picked = p; break; }
    }
    if (!picked) {
      // elegir el de menor intersección
      let best = null, bestOverlap = 1e9;
      for (const p of patterns) {
        const overlap = p.seams.filter((x) => prevSeams.has(x)).length;
        if (overlap < bestOverlap) { bestOverlap = overlap; best = p; }
      }
      picked = best || patterns[0];
    }
    chosen.push(picked);
    prevSeams = new Set(picked.seams);
  }
  return chosen;
}

function computeBOM(rows) {
  const bom = {};
  for (const r of rows) {
    for (const seg of r.segments) {
      const k = `${seg.role}:${seg.size}`;
      bom[k] = (bom[k] || 0) + 1;
    }
  }
  return bom;
}

function calcPusherRows(numRows, usePushers, mode, stepRows, offsetRows, stepIn, offsetIn, pitch){
  const P = new Set();
  if (!usePushers) return P;
  let start = 1, step = 1;
  if (mode === 'rows'){
    step = Math.max(1, Math.round(stepRows || 1));
    start = 1 + Math.max(0, Math.round(offsetRows || 0));
  } else {
    step = Math.max(1, Math.round(((stepIn || pitch)) / pitch));
    start = 1 + Math.max(0, Math.round(((offsetIn || 0)) / pitch));
  }
  for (let r = start; r <= numRows; r += step) P.add(r);
  return P;
}

function layoutRowsWithFlights(numRows, patternsNormal, patternsFlight, pusherRows){
  if (!patternsNormal.length && !patternsFlight.length) return [];
  const chosen = [];
  let prevSeams = new Set();
  for (let i = 1; i <= numRows; i++){
    const useFlight = pusherRows.has(i);
    const cand = (useFlight && patternsFlight.length) ? patternsFlight : patternsNormal;
    let picked = null;
    for (const p of cand){
      const overlap = p.seams.reduce((acc, s) => acc + (prevSeams.has(s) ? 1 : 0), 0);
      if (overlap === 0){ picked = p; break; }
    }
    if (!picked){
      let best = null, bestOverlap = 1e9;
      for (const p of cand){
        const ov = p.seams.reduce((acc, s) => acc + (prevSeams.has(s) ? 1 : 0), 0);
        if (ov < bestOverlap){ bestOverlap = ov; best = p; }
      }
      picked = best || (cand[0] || patternsNormal[0]);
    }
    let segments = picked.segments;
    if (useFlight){
      segments = picked.segments.map(seg => {
        if (seg.role === 'L') return {...seg, role:'LF'};
        if (seg.role === 'R') return {...seg, role:'RF'};
        return {...seg, role:'IF'};
      });
    }
    const row = { ...picked, segments };
    chosen.push(row);
    prevSeams = new Set(row.seams);
  }
  return chosen;
}

// --- TagSelect (selector tipo "tags" con desplegable y opciones que desaparecen si ya están elegidas) ---
function TagSelect({ label, options, selected, onChange, format }) {
  const [open, setOpen] = useState(false);
  const available = options.filter(v => !selected.has(v));
  function add(val){ const n = new Set(selected); n.add(val); onChange(n); }
  function remove(val){ const n = new Set(selected); n.delete(val); onChange(n); }
  return (
    <div>
      <div className="flex flex-wrap gap-2 mb-2">
        {Array.from(selected).sort((a,b)=>b-a).map(val => (
          <span key={val} className="inline-flex items-center gap-1 px-2 py-1 bg-gray-100 border rounded-full text-xs">
            {format(val)}
            <button type="button" onClick={()=>remove(val)} className="ml-1 text-gray-500 hover:text-black">×</button>
          </span>
        ))}
        {selected.size === 0 && <span className="text-xs text-gray-400">— vacío —</span>}
      </div>
      <div className="relative">
        <button type="button" className="px-2 py-1 border rounded-lg text-sm" onClick={()=>setOpen(o=>!o)}>
          {label} ▾
        </button>
        {open && (
          <div className="absolute z-10 mt-1 w-64 max-h-48 overflow-auto bg-white border rounded shadow">
            {available.length === 0 ? (
              <div className="p-2 text-xs text-gray-400">Sin opciones</div>
            ) : (
              available.map(val => (
                <button key={val} type="button" onClick={()=>add(val)} className="w-full text-left px-2 py-1 hover:bg-gray-50 text-sm">
                  {format(val)}
                </button>
              ))
            )}
          </div>
        )}
      </div>
    </div>
  );
}

  }

  const warnings = [];
  if (patternsNormal.length === 0) warnings.push("No hay patrones posibles con el ancho y los módulos seleccionados.");
  if (uniqueSeamSets < 2 && numRows > 1) warnings.push("Solo hay 1 set de juntas; puede haber filas consecutivas con juntas coincidentes en casos extremos.");
  if (usePushers && patternsFlight.length === 0) warnings.push("Empujadores activos, pero no hay patrones de empujadores con los tamaños seleccionados.");
  if (usePushers && pusherRows.size === 0) warnings.push("Empujadores activos, pero la configuración de paso/offset no genera ninguna fila.");

  function formatSize(sz){ return `${sz}L (${(sz*inc).toFixed(2)}` + '"' + `)`; }

  return (
    <div className="p-6 space-y-6">
      <h1 className="text-2xl font-bold">Calculadora de módulos de banda – Canvas</h1>
      <p className="text-sm text-gray-600">Reglas: orillas a extremos; priorizar menos piezas y módulos grandes; evitar alineación vertical de juntas; patrón ladrillo si es necesario; dos orillas juntas solo si su suma = ancho total.</p>

      <div className="grid lg:grid-cols-3 gap-4">
        <div className="p-4 rounded-2xl shadow bg-white space-y-3">
          <h2 className="font-semibold">Serie / Geometría</h2>
          <div className="grid grid-cols-2 gap-2 items-center">
            <label className="text-sm">Pitch (in)</label>
            <input type="number" step="0.01" value={pitch} onChange={e=>setPitch(parseFloat(e.target.value||"0")||0)} className="border rounded px-2 py-1"/>
            <label className="text-sm">Incremento por link (in)</label>
            <input type="number" step="0.01" value={inc} onChange={e=>setInc(parseFloat(e.target.value||"0")||0)} className="border rounded px-2 py-1"/>
          </div>
          <h2 className="font-semibold mt-4">Parámetros</h2>
          <div className="grid grid-cols-2 gap-2 items-center">
            <label className="text-sm">Ancho (in)</label>
            <input type="number" step="0.01" value={widthIn} onChange={e=>setWidthIn(parseFloat(e.target.value||"0")||0)} className="border rounded px-2 py-1"/>
            <label className="text-sm">Largo (in)</label>
            <input type="number" step="0.01" value={lengthIn} onChange={e=>setLengthIn(parseFloat(e.target.value||"0")||0)} className="border rounded px-2 py-1"/>
            <label className="text-sm">Redondeo</label>
            <select value={rounding} onChange={e=>setRounding(e.target.value)} className="border rounded px-2 py-1">
              <option value="round">round</option>
              <option value="floor">floor</option>
              <option value="ceil">ceil</option>
            </select>
            <label className="text-sm">px por link</label>
            <input type="number" value={pxPerLink} onChange={e=>setPxPerLink(parseInt(e.target.value||"1"))} className="border rounded px-2 py-1"/>
          </div>

          <div className="mt-4">
            <h3 className="font-semibold">Orillas disponibles (links)</h3>
            <TagSelect label="Agregar orilla" options={allSizes} selected={edgeSel} onChange={setEdgeSel} format={formatSize} />
          </div>
          <div className="mt-2">
            <h3 className="font-semibold">Internos disponibles (links)</h3>
            <TagSelect label="Agregar interno" options={allSizes} selected={intSel} onChange={setIntSel} format={formatSize} />
          </div>

          <div className="mt-4">
            <h3 className="font-semibold">Empujadores (Flights/Friction Top)</h3>
            <label className="text-sm inline-flex items-center gap-2">
              <input type="checkbox" checked={usePushers} onChange={e=>setUsePushers(e.target.checked)} />
              Activar empujadores
            </label>
            {usePushers && (
              <>
                <div className="grid grid-cols-2 gap-2 items-center mt-2">
                  <label className="text-sm">Modo</label>
                  <select value={pusherMode} onChange={e=>setPusherMode(e.target.value)} className="border rounded px-2 py-1">
                    <option value="rows">por filas</option>
                    <option value="distance">por distancia (in)</option>
                  </select>
                  {pusherMode === 'rows' ? (
                    <>
                      <label className="text-sm">Paso (filas)</label>
                      <input type="number" min="1" value={pusherStepRows} onChange={e=>setPusherStepRows(parseInt(e.target.value||"1"))} className="border rounded px-2 py-1"/>
                      <label className="text-sm">Offset (filas)</label>
                      <input type="number" min="0" value={pusherOffsetRows} onChange={e=>setPusherOffsetRows(parseInt(e.target.value||"0"))} className="border rounded px-2 py-1"/>
                    </>
                  ) : (
                    <>
                      <label className="text-sm">Paso (in)</label>
                      <input type="number" step="0.01" min="0.01" value={pusherStepIn} onChange={e=>setPusherStepIn(parseFloat(e.target.value||"0"))} className="border rounded px-2 py-1"/>
                      <label className="text-sm">Offset (in)</label>
                      <input type="number" step="0.01" min="0" value={pusherOffsetIn} onChange={e=>setPusherOffsetIn(parseFloat(e.target.value||"0"))} className="border rounded px-2 py-1"/>
                    </>
                  )}
                </div>
                <div className="mt-2">
                  <h4 className="font-medium text-sm">Orillas empujador (Flights/Friction Top) (links)</h4>
                  <TagSelect label="Agregar orilla empujador" options={allSizes} selected={edgeFlightSel} onChange={setEdgeFlightSel} format={formatSize} />
                </div>
                <div className="mt-2">
                  <h4 className="font-medium text-sm">Internos empujador (Flights/Friction Top) (links)</h4>
                  <TagSelect label="Agregar interno empujador" options={allSizes} selected={intFlightSel} onChange={setIntFlightSel} format={formatSize} />
                </div>
              </>
            )}
          </div>

          <div className="mt-4 flex gap-2">
            <button onClick={()=>setCalcTick(t=>t+1)} className="px-3 py-1 border rounded-lg bg-black/80 text-white">Calcular</button>
            <button onClick={exportBOM} className="px-3 py-1 border rounded-lg">Exportar BOM</button>
            <button onClick={exportPlanJSON} className="px-3 py-1 border rounded-lg">Exportar JSON</button>
            <button onClick={exportSeamsCSV} className="px-3 py-1 border rounded-lg">Exportar Juntas</button>
            <button onClick={runSelfTests} className="px-3 py-1 border rounded-lg">Ejecutar tests</button>
          </div>
        </div>

        <div className="p-4 rounded-2xl shadow bg-white space-y-3">
          <h2 className="font-semibold">Resumen</h2>
          <ul className="text-sm list-disc pl-5 space-y-1">
            <li>Ancho en links: <b>{widthLinks}</b></li>
            <li>Ancho real: <b>{widthRealIn.toFixed(3)}"</b> ({widthLinks}L × {inc}"/L)</li>
            <li>Filas: <b>{numRows}</b></li>
            <li>Patrones generados: <b>{patternsNormal.length}</b> (normal) + <b>{patternsFlight.length}</b> (empujador)</li>
            <li>Sets de juntas únicos: <b>{uniqueSeamSets}</b></li>
            <li>Empujadores: <b>{usePushers ? (pusherRows.size + ' filas') : 'desactivado'}</b></li>
          </ul>
          {warnings.length > 0 && (
            <div className="mt-2 p-2 rounded bg-yellow-50 border border-yellow-200 text-yellow-900 text-sm">
              <b>Atención:</b>
              <ul className="list-disc pl-5">
                {warnings.map((w, i) => <li key={i}>{w}</li>)}
              </ul>
            </div>
          )}

          {testReport && (
            <div className="mt-2 p-2 rounded bg-blue-50 border border-blue-200 text-blue-900 text-sm">
              <b>Resultados de tests:</b>
              <ul className="list-disc pl-5">
                {testReport.map((t, i) => (
                  <li key={i}><b>{t.pass ? '✔' : '✖'}</b> {t.name} — {t.detail}</li>
                ))}
              </ul>
            </div>
          )}

          <h3 className="font-semibold mt-3">BOM</h3>
          <table className="w-full text-sm border border-gray-200">
            <thead>
              <tr className="bg-gray-50"><th className="border p-1">Módulo (rol:tamaño)</th><th className="border p-1">Cantidad</th></tr>
            </thead>
            <tbody>
              {Object.entries(bom).map(([k,v]) => (
                <tr key={k}><td className="border p-1">{k}</td><td className="border p-1 text-right">{v}</td></tr>
              ))}
            </tbody>
          </table>
        </div>

        <div className="p-4 rounded-2xl shadow bg-white space-y-3 col-span-1 lg:col-span-1">
          <h2 className="font-semibold">Juntas por fila</h2>
          <div className="max-h-64 overflow-auto border rounded p-2 text-sm">
            {rows.map((r, idx) => (
              <div key={idx} className="flex items-center gap-2"><span className="w-14 text-gray-500">{idx+1}{pusherRows.has(idx+1) ? " •E" : ""}.</span><span>{[...r.seams].sort((a,b)=>a-b).join(" ") || "(sin juntas internas)"}</span></div>
            ))}
          </div>
        </div>
      </div>

      <div className="p-4 rounded-2xl shadow bg-white">
        <h2 className="font-semibold mb-2">Diagrama</h2>
        <div className="overflow-auto border rounded">
          <svg width={svgWidth} height={svgHeight} viewBox={`0 0 ${svgWidth} ${svgHeight}`} xmlns="http://www.w3.org/2000/svg">
            <rect x={0} y={0} width={svgWidth} height={svgHeight} fill="#fff" />
            {rows.map((r, rowIdx) => {
              let x = 0; const y = rowGap + rowIdx * (rowH + rowGap);
              return (
                <g key={rowIdx}>
                  {r.segments.map((seg, i) => {
                    const w = seg.size * pxPerLink;
                    const rect = (
                      <rect key={i} x={x} y={y} width={w} height={rowH} fill={segFill(seg.role, seg.size)} stroke="#111" strokeWidth={1}/>
                    );
                    x += w;
                    return rect;
                  })}
                </g>
              );
            })}
          </svg>
        </div>
      </div>

      <p className="text-xs text-gray-500">Tip: ajusta <i>px por link</i> para visualizar anchos grandes sin scroll horizontal excesivo.</p>
    </div>
  );
}
